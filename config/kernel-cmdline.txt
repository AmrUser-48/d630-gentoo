# D630 Gentoo kernel command-line contract.
#
# Preferred three-file directory boot:
#   /d630-gentoo/kernel
#   /d630-gentoo/initramfs
#   /d630-gentoo/squashfs
#
# Recommended GRUB entry passes the filesystem UUID of the partition containing
# the kernel. With no UUID argument, initramfs searches supported partitions.
# The normal folder boot requires no custom arguments; 'quiet' is optional.
d630.boot_uuid=<filesystem-uuid-containing-d630-gentoo>
quiet
#
# Optional override if the SquashFS filename/location changes:
d630.squashfs=/d630-gentoo/squashfs
#
# Folder boot uses a temporary tmpfs OverlayFS upper layer: changes vanish on
# reboot. Do not silently use NTFS as OverlayFS upperdir. Persistent folder boot
# can be added later with a separate ext4 data image on a writable partition.
#
# Legacy ISO boot remains supported for the existing ISO layout:
d630.iso_uuid=<filesystem-uuid-containing-iso>
d630.iso_path=/path/without-spaces/d630-gentoo.iso
d630.persistence=auto
d630.datafile=gentoo-data.img
d630.rootfs=/LiveOS/rootfs.squashfs
#
# Temporary ISO session:
d630.persistence=volatile
#
# Verbose early boot:
d630.debug=1 loglevel=7 ignore_loglevel
